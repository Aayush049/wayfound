89:   /// Sorts client-side without composite index:
90:   /// - Excludes archived trips

100:               .map((doc) => Trip.fromFirestore(doc))
101:               .where((t) => !t.archived)
102:               .toList();
103: 
104:           return sortTripsClientSide(trips);
105:         });
106:   }

108:   /// Client-side sorting helper for trips list.
109:   static List<Trip> sortTripsClientSide(List<Trip> trips) {
110:     final today = DateTime.now();
111:     final midnightToday = DateTime(today.year, today.month, today.day);
112: 
113:     final activeTrips = <Trip>[];
114:     final pastTrips = <Trip>[];
115: 
116:     for (final trip in trips) {
117:       final endDay = DateTime(
118:         trip.endDate.year,
119:         trip.endDate.month,
120:         trip.endDate.day,
121:       );
122:       if (endDay.isBefore(midnightToday)) {
123:         pastTrips.add(trip);
124:       } else {
125:         activeTrips.add(trip);
126:       }
127:     }
128: 
129:     // Active trips by startDate ascending
130:     activeTrips.sort((a, b) => a.startDate.compareTo(b.startDate));
131: 
132:     // Past trips by startDate descending
133:     pastTrips.sort((a, b) => b.startDate.compareTo(a.startDate));
134: 
135:     return [...activeTrips, ...pastTrips];
136:   }
